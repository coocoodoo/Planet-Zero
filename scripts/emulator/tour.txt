# Cardputer Sky tour on the emulated Cardputer Zero.
# Every action goes through the device's own key and gyroscope paths.
# Run: cargo run --release --no-default-features --features cardputer --bin flight-club-emulator

fresh
shot 00-menu
start
wait 2
expect page playing
expect location planet
expect imu imu
shot 01-arrival
bench 3

# gyroscope piloting: tilt right to bank and turn, nose up to climb
tilt 22 0 0.5
wait 2.5

press 4
wait 0.6

wait 0.5

# walk and scan

press 4

down 5
